using Crucible;
using Crucible.AI;

namespace Crucible.Scripts
{
    [NodeDescription("{{NAME}}", "{{DESCRIPTION}}", "{{CATEGORY}}", "{{ID}}")]
    public sealed class {{CLASS}} : Modifier
    {
        // Public fields show in the graph window, like an Action's.

        protected override Status OnStart()
        {

        }

        // Run the one child and decide what its result means (invert it, repeat it...).

        {
            return Child.Tick(DeltaTime);
        }

        protected override void OnEnd()
        {
        }
    }
}
